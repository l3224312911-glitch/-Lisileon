    <script>
        // 计数器
        let refuseCount = 0;
        const maxRefuseGrowth = 7;
        
        // 获取DOM元素
        const agreeBtn = document.getElementById('agree-btn');
        const refuseBtn = document.getElementById('refuse-btn');
        const screen1 = document.getElementById('screen-1');
        const screen2 = document.getElementById('screen-2');
        const screen3 = document.getElementById('screen-3');
        const reactionContent = document.getElementById('reaction-content');
        const bgm = document.getElementById('bgm');
        const musicBtn = document.getElementById('music-btn');

        bgm.volume = 0.5;

        // --- 音乐控制逻辑 ---
        
        // 手动点击按钮播放
        function playMusicManually() {
            bgm.play().then(() => {
                musicBtn.innerText = "🎵 音乐播放中...";
                musicBtn.style.opacity = "0.7"; // 播放后让按钮变淡一点
            }).catch(e => {
                console.log("播放失败");
            });
        }

        // 尝试自动播放（备用）
        function forcePlayMusic() {
            if (bgm.paused) {
                bgm.play().then(() => {
                    musicBtn.style.display = 'none'; // 如果自动播放成功，隐藏手动按钮
                }).catch(e => {
                    // 自动播放失败，保持按钮显示
                });
            }
        }

        // 1. 页面一加载就尝试
        window.onload = forcePlayMusic;

        // 2. 微信兼容
        document.addEventListener("WeixinJSBridgeReady", function () {
            forcePlayMusic();
        }, false);

        // 3. 全局点击尝试（如果用户没点按钮点了别的地方）
        document.body.addEventListener('touchstart', forcePlayMusic, { once: true });
        document.body.addEventListener('click', forcePlayMusic, { once: true });

        // --- 游戏逻辑 ---

        function clickRefuse() {
            refuseCount++;
            refuseBtn.classList.add('shake');
            setTimeout(() => refuseBtn.classList.remove('shake'), 500);

            if (refuseCount <= maxRefuseGrowth) {
                let currentFontSize = 16 + (refuseCount * 8);
                let currentPadding = 12 + (refuseCount * 5);
                agreeBtn.style.fontSize = currentFontSize + 'px';
                agreeBtn.style.padding = `${currentPadding}px ${currentPadding * 2}px`;
            } else {
                refuseBtn.innerText = "点不到了吧";
            }
        }

        function clickAgree() {
            // 确保进入下一页时音乐也是播放的
            if(bgm.paused) { bgm.play(); }

            let text = "";
            if (refuseCount === 0) {
                text = "哇哦，我真的没想到你会点同意，感动+10086";
            } else if (refuseCount >= 1 && refuseCount <= 2) {
                text = "嘻嘻，你能同意我好开心！";
            } else if (refuseCount === 3) {
                text = "虽然······拒绝我三次，但我还是很开心你能同意。";
            } else if (refuseCount === 4) {
                text = "啊呀，还是同意了吗，嘻嘻，我好开心！！！";
            } else {
                text = "呜呜呜，我的心脏好痛，特别特别痛，我的眼泪止不住的往下流···呜呜呜······但是还是谢谢你能同意，我依然很高兴。";
            }
            reactionContent.innerText = text;
            screen1.classList.add('hidden');
            screen2.classList.remove('hidden');
        }

        function goToFinal() {
            screen2.classList.add('hidden');
            screen3.classList.remove('hidden');
        }

        function resetGame() {
            refuseCount = 0;
            agreeBtn.style.fontSize = '16px';
            agreeBtn.style.padding = '12px 30px';
            refuseBtn.innerText = '我拒绝';
            screen3.classList.add('hidden');
            screen2.classList.add('hidden'); 
            screen1.classList.remove('hidden');
            
            // 重置时如果音乐没响，显示按钮
            if(bgm.paused) {
                musicBtn.style.display = 'flex';
                musicBtn.innerText = "🎵 点击此处以播放音乐";
                musicBtn.style.opacity = "1";
            }
        }
    </script>
